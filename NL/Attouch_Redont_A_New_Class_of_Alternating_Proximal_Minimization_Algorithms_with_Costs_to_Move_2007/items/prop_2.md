```lean
    (hk_lsc : LowerSemicontinuous (fun (p : Y × Y) => k p.1 p.2))

    (hk_nonneg : ∀ y y' : Y, k y y' ≥ 0)
    (hc_nonneg : ∀ x : X, ∀ y : Y, c x y ≥ 0)
    (hα_nonneg : ∀ n, α n ≥ 0) (hβ_nonneg : ∀ n, β n ≥ 0)
    (hμ_nonneg : ∀ n, μ n ≥ 0) (hν_nonneg : ∀ n, ν n ≥ 0)
    (β_inf : ℝ) (hβ : Tendsto β atTop (𝓝 β_inf))
    (μ_inf : ℝ) (hμ : Tendsto μ atTop (𝓝 μ_inf))
```

## Statement explanation
The declaration `proposition_2_2` states an asymptotic equilibrium property for a sequence generated by a `CA2` algorithm. The ambient data are two topological spaces \(X\) and \(Y\), real-valued functions
\[
f:X\to \mathbb R,\qquad g:Y\to \mathbb R,\qquad
h:X\times X\to \mathbb R,\qquad k:Y\times Y\to \mathbb R,\qquad
c:X\times Y\to \mathbb R,
\]
and coefficient sequences
\[
\alpha,\beta,\mu,\nu:\mathbb N\to \mathbb R.
\]
The object `alg : CA2 X Y f g h k c α β μ ν` supplies the algorithm state sequences \(x_n=\text{alg.x}(n)\) and \(y_n=\text{alg.y}(n)\). The theorem assumes that these sequences converge:
\[
x_n\to \bar x,\qquad y_n\to \bar y .
\]

The remaining hypotheses encode regularity, nonnegativity, and limiting behavior of the data:

- \(f\) and \(g\) are lower semicontinuous and bounded below.
- \(h\), \(k\), and \(c\) are lower semicontinuous as functions on the product spaces \(X\times X\), \(Y\times Y\), and \(X\times Y\), respectively.
- \(h\), \(k\), and \(c\) are separately continuous.
- The costs are nonnegative:
  \[
  h(x,x')\ge 0,\qquad k(y,y')\ge 0,\qquad c(x,y)\ge 0 .
  \]
- The coefficient sequences are nonnegative:
  \[
  \alpha_n\ge 0,\quad \beta_n\ge 0,\quad \mu_n\ge 0,\quad \nu_n\ge 0 .
  \]
- \(\beta\) and \(\mu\) converge to real numbers \(\beta_\infty\) and \(\mu_\infty\).
- The diagonal costs vanish:
  \[
  h(x,x)=0,\qquad k(y,y)=0 .
  \]
- The numbers \(\bar\alpha\) and \(\bar\nu\) bound the limsups of \(\alpha\) and \(\nu\) in the following eventual sense:
  \[
  \forall \varepsilon>0,\ \text{eventually } \alpha_n\le \bar\alpha+\varepsilon,
  \qquad
  \forall \varepsilon>0,\ \text{eventually } \nu_n\le \bar\nu+\varepsilon .
  \]
- The final coefficients satisfy
  \[
  \alpha_{\mathrm{val}}\ge \bar\alpha,\qquad \nu_{\mathrm{val}}\ge \bar\nu .
  \]

The conclusion is
\[
\text{IsINE }f\,g\,h\,k\,c\,\alpha_{\mathrm{val}}\,\beta_\infty\,\mu_\infty\,\nu_{\mathrm{val}}\,\bar x\,\bar y .
\]
From the final `refine ⟨?_, ?_⟩`, this conclusion is the conjunction of the following two universal inequalities:
\[
\forall \xi\in X,\qquad
f(\bar x)+\beta_\infty c(\bar x,\bar y)+\alpha_{\mathrm{val}}h(\bar x,\bar x)
\le
f(\xi)+\beta_\infty c(\xi,\bar y)+\alpha_{\mathrm{val}}h(\bar x,\xi),
\]
and
\[
\forall \eta\in Y,\qquad
g(\bar y)+\mu_\infty c(\bar x,\bar y)+\nu_{\mathrm{val}}k(\bar y,\bar y)
\le
g(\eta)+\mu_\infty c(\bar x,\eta)+\nu_{\mathrm{val}}k(\bar y,\eta).
\]
Since \(h(\bar x,\bar x)=0\) and \(k(\bar y,\bar y)=0\), the diagonal terms vanish mathematically, but the definition of `IsINE` used here keeps them explicitly. In ordinary language, the theorem says that if a `CA2` algorithm converges and the costs and coefficients have the stated regularity and nonnegativity, then the limit point \((\bar x,\bar y)\) satisfies the two coupled equilibrium inequalities with coefficients \(\alpha_{\mathrm{val}},\beta_\infty,\mu_\infty,\nu_{\mathrm{val}}\). These inequalities express that no unilateral change \(\xi\) in the \(x\)-variable or \(\eta\) in the \(y\)-variable can improve the limiting objective beyond the value at \((\bar x,\bar y)\) plus the corresponding penalty terms.

```lean
      ν n * k (alg.y n) η ≤ (νbar + δ) * k (alg.y n) η :=
        mul_le_mul_of_nonneg_right hn_ν h_nonneg_k
      _ ≤ (νbar + δ) * (k ybar η + δ) :=
        mul_le_mul_of_nonneg_left hn_k.le (by nlinarith [h_νbar_nonneg])
      _ < νbar * k ybar η + ε := h_bound

  -- ============================================================
  -- RHS CONVERGENCE (excluding α_n terms which don't converge)
  -- β_n * c(ξ, y_n) → β_inf * c(ξ, ybar) (converges)
  -- ============================================================
  have hRHS_x_partial (ξ : X) :
      Tendsto (fun n : ℕ => f ξ + β n * c ξ (alg.y n)) atTop
      (𝓝 (f ξ + β_inf * c ξ ybar)) := by
    have hcξ : Tendsto (fun n : ℕ => c ξ (alg.y n)) atTop (𝓝 (c ξ ybar)) :=
      ((hc_sep2 ξ).continuousAt.tendsto).comp hy
    have hβcξ : Tendsto (fun n : ℕ => β n * c ξ (alg.y n)) atTop (𝓝 (β_inf * c ξ ybar)) :=
      Filter.Tendsto.mul hβ hcξ
    simpa using ((tendsto_const_nhds : Tendsto (fun _ : ℕ => f ξ) atTop _).add hβcξ)

  -- ============================================================
  -- MAIN PROOF: First INE conjunct (x-coordinate)
  -- f xbar + β_inf * c xbar ybar ≤ f ξ + β_inf * c ξ ybar + α_val * h xbar ξ
  -- ============================================================
  have h_xine : ∀ (ξ : X), f xbar + β_inf * c xbar ybar + α_val * h xbar xbar ≤
      f ξ + β_inf * c ξ ybar + α_val * h xbar ξ := by
    intro ξ
    have h_hbar_diag_zero : h xbar xbar = 0 := h_diag_h xbar
    -- The goal simplifies to: f xbar + β_inf * c xbar ybar ≤ f ξ + β_inf * c ξ ybar + α_val * h xbar ξ
    have h_target : f xbar + β_inf * c xbar ybar + α_val * h xbar xbar =
        f xbar + β_inf * c xbar ybar := by
      simp [h_hbar_diag_zero]
    rw [h_target]
    refine le_of_forall_pos_lt_add ?_
    intro ε hε
    -- Write ε = ε₁ + ε₂ + ε₃ (three sources of error)
    set ε₀ := ε / 5 with hε₀
    have hε₀pos : ε₀ > 0 := by linarith
    -- (1) f(xbar) - ε₀ < f(x_{n+1}) eventually
    have h_f_ev : ∀ᶠ n in atTop, f xbar - ε₀ < f (alg.x (n+1)) :=
      h_fx_lower ε₀ hε₀pos
    -- (2) β_inf * c(xbar, ybar) - ε₀ < β_n * c(x_{n+1}, y_n) eventually (product lemma)
    have h_βc_ev : ∀ᶠ n in atTop, β_inf * c xbar ybar - ε₀ < β n * c (alg.x (n+1)) (alg.y n) :=
      h_prod_ev_lower hε₀pos hβ_nonneg (λ n => hc_nonneg _ _) hβ h_cbar_nonneg h_c_lower
    -- (3) α_n * h(x_n, x_{n+1}) ≥ 0 (dropped from LHS)
    have h_αh_nonneg (n : ℕ) : 0 ≤ α n * h (alg.x n) (alg.x (n+1)) :=
      mul_nonneg (hα_nonneg n) (hh_nonneg _ _)
    -- (4) RHS partial bound: f(ξ) + β_n * c(ξ, y_n) < f(ξ) + β_inf * c(ξ, ybar) + ε₀
    have hRHS_partial_ev : ∀ᶠ n in atTop,
        f ξ + β n * c ξ (alg.y n) < f ξ + β_inf * c ξ ybar + ε₀ := by
      have ht := hRHS_x_partial ξ
      have hball : Metric.ball (f ξ + β_inf * c ξ ybar : ℝ) ε₀ ∈ 𝓝 (f ξ + β_inf * c ξ ybar : ℝ) :=
        Metric.ball_mem_nhds _ hε₀pos
      filter_upwards [ht.eventually hball] with n hn
      have hdist : dist (f ξ + β n * c ξ (alg.y n)) (f ξ + β_inf * c ξ ybar) < ε₀ := hn
      rw [Real.dist_eq] at hdist
      have h_abs := abs_lt.mp hdist
      linarith
    -- (5) α_n * h(x_n, ξ) < αbar * h(xbar, ξ) + ε₀ eventually (upper bound lemma)
    have h_αh_upper_ev : ∀ᶠ n in atTop, α n * h (alg.x n) ξ < αbar * h xbar ξ + ε₀ :=
      h_αh_upper_bound ξ ε₀ hε₀pos
    -- Gather all eventual bounds
    have h_all : ∀ᶠ n in atTop,
        f xbar - ε₀ < f (alg.x (n+1)) ∧
        β_inf * c xbar ybar - ε₀ < β n * c (alg.x (n+1)) (alg.y n) ∧
        (f (alg.x (n+1)) + α n * h (alg.x n) (alg.x (n+1)) + β n * c (alg.x (n+1)) (alg.y n) ≤
         f ξ + α n * h (alg.x n) ξ + β n * c ξ (alg.y n)) ∧
        α n * h (alg.x n) ξ < αbar * h xbar ξ + ε₀ ∧
        f ξ + β n * c ξ (alg.y n) < f ξ + β_inf * c ξ ybar + ε₀ := by
      filter_upwards [h_f_ev, h_βc_ev, h_αh_upper_ev, hRHS_partial_ev] with
        n hf hβc hαh_upper hRHSpart
      have h_step : f (alg.x (n+1)) + α n * h (alg.x n) (alg.x (n+1)) +
          β n * c (alg.x (n+1)) (alg.y n) ≤
          f ξ + α n * h (alg.x n) ξ + β n * c ξ (alg.y n) :=
        alg.step_x n ξ
      exact ⟨hf, hβc, h_step, hαh_upper, hRHSpart⟩
    obtain ⟨n, hn⟩ := Filter.Eventually.exists h_all
    rcases hn with ⟨hf, hβc, hstep, hαh_upper, hRHSpart⟩
    -- Chain inequalities:
    -- LHS: (f(xbar) - ε₀) + (β_inf*c(xbar,ybar) - ε₀) < LHS_n
    -- LHS_n ≤ RHS_n (from algorithm)
    -- RHS_n = f(ξ) + α_n*h(x_n,ξ) + β_n*c(ξ,y_n) < f(ξ) + (αbar*h(xbar,ξ) + ε₀) + (f(ξ) + β_inf*c(ξ,ybar) + ε₀ - f(ξ))
    -- Wait, RHS_n = f(ξ) + α_n*h(x_n,ξ) + β_n*c(ξ,y_n)
    -- < f(ξ) + (αbar*h(xbar,ξ) + ε₀) + β_n*c(ξ,y_n) [using hαh_upper]
    -- Let's do: RHS_n < f(ξ) + αbar*h(xbar,ξ) + ε₀ + β_n*c(ξ,y_n)
    -- And β_n*c(ξ,y_n) < f(ξ) + β_inf*c(ξ,ybar) + ε₀ - f(ξ) = β_inf*c(ξ,ybar) + ε₀ [from hRHSpart]
    -- So RHS_n < f(ξ) + αbar*h(xbar,ξ) + β_inf*c(ξ,ybar) + 2ε₀
    -- Now use α_val ≥ αbar: αbar*h(xbar,ξ) ≤ α_val*h(xbar,ξ)
    -- RHS_n < f(ξ) + α_val*h(xbar,ξ) + β_inf*c(ξ,ybar) + 2ε₀
    -- And LHS_n ≤ RHS_n from algorithm, combined with LHS lower bound:
    -- f(xbar) + β_inf*c(xbar,ybar) - 2ε₀ < LHS_n + α_n*h(x_n,x_{n+1}) [hmm, we need to be careful]
    -- Actually: LHS_n = f(x_{n+1}) + α_n*h(x_n, x_{n+1}) + β_n*c(x_{n+1},y_n)
    -- Since α_n*h(x_n, x_{n+1}) ≥ 0:
    -- f(x_{n+1}) + β_n*c(x_{n+1},y_n) ≤ LHS_n
    -- But hstep gives: LHS_n ≤ RHS_n
    -- Lower bound on f + βc: (f(xbar) - ε₀) + (β_inf*c(xbar,ybar) - ε₀) < f(x_{n+1}) + β_n*c(x_{n+1},y_n)
    -- So: f(xbar) + β_inf*c(xbar,ybar) - 2ε₀ < f(x_{n+1}) + β_n*c(x_{n+1},y_n) ≤ LHS_n ≤ RHS_n
    -- RHS_n = f(ξ) + α_n*h(x_n,ξ) + β_n*c(ξ,y_n)
    -- < f(ξ) + (αbar*h(xbar,ξ) + ε₀) + (β_inf*c(ξ,ybar) + ε₀)  [subtracting f(ξ) from hRHSpart]
    -- Wait: hRHSpart says f(ξ) + β_n*c(ξ,y_n) < f(ξ) + β_inf*c(ξ,ybar) + ε₀
    -- So β_n*c(ξ,y_n) < β_inf*c(ξ,ybar) + ε₀
    -- And hαh_upper says α_n*h(x_n,ξ) < αbar*h(xbar,ξ) + ε₀
    -- So RHS_n = f(ξ) + α_n*h(x_n,ξ) + β_n*c(ξ,y_n)
    -- < f(ξ) + αbar*h(xbar,ξ) + ε₀ + β_inf*c(ξ,ybar) + ε₀
    -- = f(ξ) + αbar*h(xbar,ξ) + β_inf*c(ξ,ybar) + 2ε₀
    -- Now since α_val ≥ αbar and h(xbar,ξ) ≥ 0:
    -- αbar*h(xbar,ξ) ≤ α_val*h(xbar,ξ)
    -- So RHS_n < f(ξ) + α_val*h(xbar,ξ) + β_inf*c(ξ,ybar) + 2ε₀
    -- Therefore: f(xbar) + β_inf*c(xbar,ybar) - 2ε₀ < f(ξ) + α_val*h(xbar,ξ) + β_inf*c(ξ,ybar) + 2ε₀
    -- So: f(xbar) + β_inf*c(xbar,ybar) < f(ξ) + α_val*h(xbar,ξ) + β_inf*c(ξ,ybar) + 4ε₀
    -- And 4ε₀ = 4ε/5 < ε, done.
    have hLHS_simple_lower : f xbar + β_inf * c xbar ybar - 2 * ε₀ <
        f (alg.x (n+1)) + β n * c (alg.x (n+1)) (alg.y n) := by
      nlinarith
    have hLHS_le_RHS : f (alg.x (n+1)) + β n * c (alg.x (n+1)) (alg.y n) ≤
        f ξ + α n * h (alg.x n) ξ + β n * c ξ (alg.y n) := by
      have h_nonneg_term : 0 ≤ α n * h (alg.x n) (alg.x (n+1)) := h_αh_nonneg n
      linarith
    have hRHS_upper : f ξ + α n * h (alg.x n) ξ + β n * c ξ (alg.y n) <
        f ξ + α_val * h xbar ξ + β_inf * c ξ ybar + 2 * ε₀ := by
      -- α_n*h(x_n,ξ) < αbar*h(xbar,ξ) + ε₀ ≤ α_val*h(xbar,ξ) + ε₀ (since h ≥ 0)
      have h_α_term : α n * h (alg.x n) ξ ≤ α_val * h xbar ξ + ε₀ := by
        have h_nonneg_h : 0 ≤ h xbar ξ := hh_nonneg _ _
        have : αbar * h xbar ξ ≤ α_val * h xbar ξ :=
          mul_le_mul_of_nonneg_right hα_ge h_nonneg_h
        linarith
      -- β_n*c(ξ,y_n) < β_inf*c(ξ,ybar) + ε₀ (from hRHSpart)
      -- hRHSpart: f(ξ) + β_n*c(ξ,y_n) < f(ξ) + β_inf*c(ξ,ybar) + ε₀
      -- So β_n*c(ξ,y_n) < β_inf*c(ξ,ybar) + ε₀
      nlinarith
    have h_final : f xbar + β_inf * c xbar ybar <
        f ξ + α_val * h xbar ξ + β_inf * c ξ ybar + ε := by
      nlinarith
    linarith

  -- ============================================================
  -- MAIN PROOF: Second INE conjunct (y-coordinate)
  -- g ybar + μ_inf * c xbar ybar ≤ g η + μ_inf * c xbar η + ν_val * k ybar η
  -- ============================================================
  have h_yine : ∀ (η : Y), g ybar + μ_inf * c xbar ybar + ν_val * k ybar ybar ≤
      g η + μ_inf * c xbar η + ν_val * k ybar η := by
    intro η
    have h_kbar_diag_zero : k ybar ybar = 0 := h_diag_k ybar
    -- The goal simplifies to: g ybar + μ_inf * c xbar ybar ≤ g η + μ_inf * c xbar η + ν_val * k ybar η
    have h_target : g ybar + μ_inf * c xbar ybar + ν_val * k ybar ybar =
        g ybar + μ_inf * c xbar ybar := by
      simp [h_kbar_diag_zero]
    rw [h_target]
    refine le_of_forall_pos_lt_add ?_
    intro ε hε
    set ε₀ := ε / 5 with hε₀
    have hε₀pos : ε₀ > 0 := by linarith
    -- (1) g(ybar) - ε₀ < g(y_{n+1}) eventually
    have h_g_ev : ∀ᶠ n in atTop, g ybar - ε₀ < g (alg.y (n+1)) :=
      h_gy_lower ε₀ hε₀pos
    -- (2) μ_inf * c(xbar, ybar) - ε₀ < μ_n * c(x_{n+1}, y_{n+1}) eventually (product lemma)
    have h_μc_ev : ∀ᶠ n in atTop, μ_inf * c xbar ybar - ε₀ < μ n * c (alg.x (n+1)) (alg.y (n+1)) :=
      h_prod_ev_lower hε₀pos hμ_nonneg (λ n => hc_nonneg _ _) hμ h_cbar_nonneg h_c2_lower
    -- (3) ν_n * k(y_n, y_{n+1}) ≥ 0 (dropped from LHS)
    have h_νk_nonneg (n : ℕ) : 0 ≤ ν n * k (alg.y n) (alg.y (n+1)) :=
      mul_nonneg (hν_nonneg n) (hk_nonneg _ _)
    -- (4) RHS partial bound (c(x_{n+1}, η) → c(xbar, η), μ_n → μ_inf converge)
    have hRHS_y_partial (η : Y) :
        Tendsto (fun n : ℕ => g η + μ n * c (alg.x (n+1)) η) atTop
        (𝓝 (g η + μ_inf * c xbar η)) := by
      have hcη : Tendsto (fun n : ℕ => c (alg.x (n+1)) η) atTop (𝓝 (c xbar η)) :=
        ((hc_sep1 η).continuousAt.tendsto).comp hx_shift
      have hμcη : Tendsto (fun n : ℕ => μ n * c (alg.x (n+1)) η) atTop (𝓝 (μ_inf * c xbar η)) :=
        Filter.Tendsto.mul hμ hcη
      simpa using ((tendsto_const_nhds : Tendsto (fun _ : ℕ => g η) atTop _).add hμcη)
    have hRHS_partial_ev : ∀ᶠ n in atTop,
        g η + μ n * c (alg.x (n+1)) η < g η + μ_inf * c xbar η + ε₀ := by
      have ht := hRHS_y_partial η
      have hball : Metric.ball (g η + μ_inf * c xbar η : ℝ) ε₀ ∈ 𝓝 (g η + μ_inf * c xbar η : ℝ) :=
        Metric.ball_mem_nhds _ hε₀pos
      filter_upwards [ht.eventually hball] with n hn
      have hdist : dist (g η + μ n * c (alg.x (n+1)) η) (g η + μ_inf * c xbar η) < ε₀ := hn
      rw [Real.dist_eq] at hdist
      have h_abs := abs_lt.mp hdist
      linarith
    -- (5) ν_n * k(y_n, η) < νbar * k(ybar, η) + ε₀ eventually (upper bound lemma)
    have h_νk_upper_ev : ∀ᶠ n in atTop, ν n * k (alg.y n) η < νbar * k ybar η + ε₀ :=
      h_νk_upper_bound η ε₀ hε₀pos
    -- Gather all eventual bounds
    have h_all : ∀ᶠ n in atTop,
        g ybar - ε₀ < g (alg.y (n+1)) ∧
        μ_inf * c xbar ybar - ε₀ < μ n * c (alg.x (n+1)) (alg.y (n+1)) ∧
        (g (alg.y (n+1)) + μ n * c (alg.x (n+1)) (alg.y (n+1)) + ν n * k (alg.y n) (alg.y (n+1)) ≤
         g η + μ n * c (alg.x (n+1)) η + ν n * k (alg.y n) η) ∧
        ν n * k (alg.y n) η < νbar * k ybar η + ε₀ ∧
        g η + μ n * c (alg.x (n+1)) η < g η + μ_inf * c xbar η + ε₀ := by
      filter_upwards [h_g_ev, h_μc_ev, h_νk_upper_ev, hRHS_partial_ev] with
        n hg hμc hνk_upper hRHSpart
      have h_step : g (alg.y (n+1)) + μ n * c (alg.x (n+1)) (alg.y (n+1)) +
          ν n * k (alg.y n) (alg.y (n+1)) ≤
          g η + μ n * c (alg.x (n+1)) η + ν n * k (alg.y n) η :=
        alg.step_y n η
      exact ⟨hg, hμc, h_step, hνk_upper, hRHSpart⟩
    obtain ⟨n, hn⟩ := Filter.Eventually.exists h_all
    rcases hn with ⟨hg, hμc, hstep, hνk_upper, hRHSpart⟩
    -- Chain inequalities (same pattern as x-coordinate)
    have hLHS_simple_lower : g ybar + μ_inf * c xbar ybar - 2 * ε₀ <
        g (alg.y (n+1)) + μ n * c (alg.x (n+1)) (alg.y (n+1)) := by
      nlinarith
    have hLHS_le_RHS : g (alg.y (n+1)) + μ n * c (alg.x (n+1)) (alg.y (n+1)) ≤
        g η + μ n * c (alg.x (n+1)) η + ν n * k (alg.y n) η := by
      have h_nonneg_term : 0 ≤ ν n * k (alg.y n) (alg.y (n+1)) := h_νk_nonneg n
      linarith
    have hRHS_upper : g η + μ n * c (alg.x (n+1)) η + ν n * k (alg.y n) η <
        g η + ν_val * k ybar η + μ_inf * c xbar η + 2 * ε₀ := by
      have h_ν_term : ν n * k (alg.y n) η ≤ ν_val * k ybar η + ε₀ := by
        have h_nonneg_k : 0 ≤ k ybar η := hk_nonneg _ _
        have : νbar * k ybar η ≤ ν_val * k ybar η :=
          mul_le_mul_of_nonneg_right hν_ge h_nonneg_k
        linarith
      nlinarith
    have h_final : g ybar + μ_inf * c xbar ybar <
        g η + ν_val * k ybar η + μ_inf * c xbar η + ε := by
      nlinarith
    nlinarith

  refine ⟨?_, ?_⟩
  · intro ξ
    -- Need: f xbar + β_inf * c xbar ybar + α_val * h xbar xbar ≤ f ξ + β_inf * c ξ ybar + α_val * h xbar ξ
    -- This is exactly h_xine ξ
    have h := h_xine ξ
    -- h_xine ξ gives: f xbar + β_inf * c xbar ybar + α_val * h xbar xbar ≤ f ξ + β_inf * c ξ ybar + α_val * h xbar ξ
    -- But the IsINE definition has the terms in a different order. Use ring/add_comm/add_assoc.
    simpa [add_comm, add_left_comm, add_assoc] using h
  · intro η
    have h := h_yine η
    simpa [add_comm, add_left_comm, add_assoc] using h
```

## Proof explanation
The proof is a long limiting argument. It first records several consequences of the hypotheses, then proves a product lower-bound lemma and two upper-bound lemmas, and finally assembles the two conjuncts of `IsINE`.

First, the proof shifts the convergent sequences. The lemma `tendsto_add_atTop_nat 1` states that the sequence \(n\mapsto n+1\) tends to infinity in the filter \(\text{atTop}\). Composing the convergence \(x_n\to\bar x\) with this shift gives
\[
x_{n+1}\to \bar x,
\]
and similarly
\[
y_{n+1}\to \bar y .
\]
These shifted limits are used later when the algorithm step involves the \((n+1)\)-st iterates.

Next, the separate-continuity hypotheses are decomposed. The term `IsSeparatelyContinuous h` means that for each fixed second argument the first-variable function is continuous, and for each fixed first argument the second-variable function is continuous. The proof writes `rcases hh_sep with ⟨hh_sep1, hh_sep2⟩`, and similarly for \(k\) and \(c\). In the proof, for example, `hh_sep1 ξ` is the continuity of
\[
x\mapsto h(x,\xi),
\]
and `hc_sep2 ξ` is the continuity of
\[
y\mapsto c(\xi,y).
\]

The proof then proves that the limsup bounds \(\bar\alpha\) and \(\bar\nu\) are nonnegative. For \(\bar\alpha\), suppose for contradiction that \(\bar\alpha<0\). Choose
\[
\varepsilon=-\frac{\bar\alpha}{2}>0.
\]
By the defining property of \(\bar\alpha\), eventually
\[
\alpha_n\le \bar\alpha+\varepsilon=\frac{\bar\alpha}{2}<0.
\]
Since the filter \(\text{atTop}\) on \(\mathbb N\) has an actual element satisfying an eventual property, the proof extracts some \(n\) with \(\alpha_n\le \bar\alpha/2\). But the hypothesis gives \(\alpha_n\ge0\) for every \(n\), a contradiction. Thus \(\bar\alpha\ge0\). The same argument gives \(\bar\nu\ge0\).

The limits \(\beta_\infty\) and \(\mu_\infty\) are also shown to be nonnegative. The lemma `ge_of_tendsto` says that if a sequence tends to a limit and is eventually at least a constant, then the limit is at least that constant. Since \(\beta_n\ge0\) for all \(n\), and hence eventually, and \(\beta_n\to\beta_\infty\), it follows that \(\beta_\infty\ge0\). Similarly \(\mu_\infty\ge0\).

The nonnegativity hypotheses on the costs immediately give
\[
h(\bar x,\bar x)\ge0,\qquad k(\bar y,\bar y)\ge0,\qquad c(\bar x,\bar y)\ge0.
\]

The proof then forms joint convergence of pairs. The lemma `Filter.Tendsto.prodMk_nhds` states that if \(x_{n+1}\to\bar x\) and \(y_n\to\bar y\), then
\[
(x_{n+1},y_n)\to(\bar x,\bar y)
\]
in the product topology. Similarly,
\[
(x_{n+1},y_{n+1})\to(\bar x,\bar y).
\]
This allows the lower semicontinuity of \(c\) on \(X\times Y\) to be applied to the sequences of pairs.

Lower semicontinuity of \(f\) gives the following eventual lower bound. For every \(\varepsilon>0\), since \(f(\bar x)-\varepsilon<f(\bar x)\), lower semicontinuity at \(\bar x\) gives a neighborhood of \(\bar x\) on which \(f(z)>f(\bar x)-\varepsilon\). Because \(x_{n+1}\to\bar x\), eventually
\[
f(\bar x)-\varepsilon<f(x_{n+1}).
\]
Similarly, lower semicontinuity of \(g\) gives eventually
\[
g(\bar y)-\varepsilon<g(y_{n+1}).
\]
Lower semicontinuity of \(c\) on the product gives eventually
\[
c(\bar x,\bar y)-\varepsilon<c(x_{n+1},y_n)
\]
and also
\[
c(\bar x,\bar y)-\varepsilon<c(x_{n+1},y_{n+1}),
\]
using the two product-convergence statements above.

The convergence \(\beta_n\to\beta_\infty\) gives eventual lower bounds of the form
\[
\beta_\infty-\delta<\beta_n
\]
for every \(\delta>0\): the set \(\{x:\beta_\infty-\delta<x\}\) is open and contains \(\beta_\infty\), hence is a neighborhood, and convergence eventually lands in it. The same argument gives
\[
\mu_\infty-\delta<\mu_n
\]
eventually.

The proof then establishes a product lower-bound lemma. Suppose \(a_n\ge0\), \(b_n\ge0\), \(a_n\to a_\infty\), \(b_\infty\ge0\), and for every \(\delta>0\), eventually
\[
b_\infty-\delta<b_n .
\]
Then for every \(\varepsilon>0\), eventually
\[
a_\infty b_\infty-\varepsilon<a_n b_n .
\]
The proof first notes that \(a_\infty\ge0\), by `ge_of_tendsto` applied to \(a_n\ge0\). If \(a_\infty=0\), then \(a_\infty b_\infty=0\), while \(a_n b_n\ge0\), so \(-\varepsilon<a_n b_n\). If \(a_\infty>0\), the proof sets
\[
D=a_\infty+b_\infty+1>0,\qquad
\eta=\min\left(\frac{a_\infty}{2},\frac{\varepsilon}{2D}\right)>0.
\]
Then \(\eta<a_\infty\). It proves the algebraic inequality
\[
a_\infty b_\infty-\varepsilon<(a_\infty-\eta)(b_\infty-\eta)
\]
by expanding the difference, which is
\[
\eta(a_\infty+b_\infty-\eta).
\]
The choice of \(\eta\) makes this less than \(\varepsilon\). Eventually \(a_n>a_\infty-\eta\) by convergence, and \(b_n>b_\infty-\eta\) by hypothesis. If \(b_\infty-\eta>0\), monotonicity of multiplication gives
\[
(a_\infty-\eta)(b_\infty-\eta)<a_n b_n .
\]
If \(b_\infty-\eta\le0\), then the left side is nonpositive while \(a_n b_n\ge0\), so the same strict inequality holds. Thus \(a_\infty b_\infty-\varepsilon<a_n b_n\) eventually.

This product lemma is used twice. First, with
\[
a_n=\beta_n,\qquad b_n=c(x_{n+1},y_n),\qquad a_\infty=\beta_\infty,\qquad b_\infty=c(\bar x,\bar y),
\]
it yields eventually
\[
\beta_\infty c(\bar x,\bar y)-\varepsilon
<
\beta_n c(x_{n+1},y_n).
\]
Second, with
\[
a_n=\mu_n,\qquad b_n=c(x_{n+1},y_{n+1}),\qquad a_\infty=\mu_\infty,\qquad b_\infty=c(\bar x,\bar y),
\]
it yields eventually
\[
\mu_\infty c(\bar x,\bar y)-\varepsilon
<
\mu_n c(x_{n+1},y_{n+1}).
\]

The proof next establishes an upper-bound lemma for the terms \(\alpha_n h(x_n,\xi)\). Fix \(\xi\in X\) and \(\varepsilon>0\). Since \(h(\bar x,\xi)\ge0\), and since \(h(x_n,\xi)\to h(\bar x,\xi)\) by separate continuity, the proof chooses
\[
C=\bar\alpha+h(\bar x,\xi)+1>0,\qquad
\delta=\min\left(\frac{\varepsilon}{2C},1\right)>0.
\]
It verifies
\[
(\bar\alpha+\delta)(h(\bar x,\xi)+\delta)
<
\bar\alpha h(\bar x,\xi)+\varepsilon .
\]
Indeed, expanding the left side gives
\[
\bar\alpha h(\bar x,\xi)+\delta(\bar\alpha+h(\bar x,\xi)+\delta),
\]
and the choice of \(\delta\) makes the correction term at most \(\varepsilon/2\). Eventually \(\alpha_n\le\bar\alpha+\delta\) by the defining limsup property of \(\bar\alpha\), and eventually
\[
h(x_n,\xi)<h(\bar x,\xi)+\delta
\]
by convergence \(h(x_n,\xi)\to h(\bar x,\xi)\). Since \(h(x_n,\xi)\ge0\) and \(\bar\alpha+\delta\ge0\), multiplication preserves the inequalities, giving eventually
\[
\alpha_n h(x_n,\xi)
<
\bar\alpha h(\bar x,\xi)+\varepsilon.
\]
The same argument with \(\nu\), \(k\), and \(\bar\nu\) gives eventually
\[
\nu_n k(y_n,\eta)
<
\bar\nu k(\bar y,\eta)+\varepsilon.
\]

The proof also records convergence of the partial right-hand sides. For fixed \(\xi\in X\), separate continuity of \(c\) in the second variable gives
\[
c(\xi,y_n)\to c(\xi,\bar y).
\]
Multiplying by \(\beta_n\to\beta_\infty\) gives
\[
\beta_n c(\xi,y_n)\to \beta_\infty c(\xi,\bar y),
\]
and adding the constant \(f(\xi)\) gives
\[
f(\xi)+\beta_n c(\xi,y_n)
\to
f(\xi)+\beta_\infty c(\xi,\bar y).
\]
Thus eventually this quantity is less than the limit plus \(\varepsilon_0\). Similarly, for fixed \(\eta\in Y\), separate continuity of \(c\) in the first variable gives
\[
c(x_{n+1},\eta)\to c(\bar x,\eta),
\]
and hence
\[
g(\eta)+\mu_n c(x_{n+1},\eta)
\to
g(\eta)+\mu_\infty c(\bar x,\eta),
\]
so eventually it is less than the limit plus \(\varepsilon_0\).

Now the proof proves the first `IsINE` inequality. Fix \(\xi\in X\). The goal initially contains the diagonal term \(\alpha_{\mathrm{val}}h(\bar x,\bar x)\), which is rewritten to zero using \(h(\bar x,\bar x)=0\). It is then enough to prove
\[
f(\bar x)+\beta_\infty c(\bar x,\bar y)
\le
f(\xi)+\beta_\infty c(\xi,\bar y)+\alpha_{\mathrm{val}}h(\bar x,\xi).
\]
The lemma `le_of_forall_pos_lt_add` reduces this to proving, for every \(\varepsilon>0\),
\[
f(\bar x)+\beta_\infty c(\bar x,\bar y)
<
f(\xi)+\beta_\infty c(\xi,\bar y)+\alpha_{\mathrm{val}}h(\bar x,\xi)+\varepsilon .
\]
The proof sets \(\varepsilon_0=\varepsilon/5\). It then collects an index \(n\) satisfying all the eventual inequalities simultaneously. At such an \(n\), lower semicontinuity and the product lemma give
\[
f(\bar x)+\beta_\infty c(\bar x,\bar y)-2\varepsilon_0
<
f(x_{n+1})+\beta_n c(x_{n+1},y_n).
\]
The algorithm step `alg.step_x n ξ` gives
\[
f(x_{n+1})+\alpha_n h(x_n,x_{n+1})+\beta_n c(x_{n+1},y_n)
\le
f(\xi)+\alpha_n h(x_n,\xi)+\beta_n c(\xi,y_n).
\]
Since \(\alpha_n h(x_n,x_{n+1})\ge0\), the nonnegative term can be dropped:
\[
f(x_{n+1})+\beta_n c(x_{n+1},y_n)
\le
f(\xi)+\alpha_n h(x_n,\xi)+\beta_n c(\xi,y_n).
\]
Using the upper bound for \(\alpha_n h(x_n,\xi)\), together with \(\bar\alpha\le\alpha_{\mathrm{val}}\) and \(h(\bar x,\xi)\ge0\), and using the convergence of the partial right-hand side, one obtains
\[
f(\xi)+\alpha_n h(x_n,\xi)+\beta_n c(\xi,y_n)
<
f(\xi)+\alpha_{\mathrm{val}}h(\bar x,\xi)
+\beta_\infty c(\xi,\bar y)+2\varepsilon_0.
\]
Combining the lower and upper chains yields
\[
f(\bar x)+\beta_\infty c(\bar x,\bar y)-2\varepsilon_0
<
f(\xi)+\alpha_{\mathrm{val}}h(\bar x,\xi)
+\beta_\infty c(\xi,\bar y)+2\varepsilon_0.
\]
Therefore
\[
f(\bar x)+\beta_\infty c(\bar x,\bar y)
<
f(\xi)+\alpha_{\mathrm{val}}h(\bar x,\xi)
+\beta_\infty c(\xi,\bar y)+4\varepsilon_0.
\]
Since \(4\varepsilon_0=4\varepsilon/5<\varepsilon\), the desired strict inequality with \(\varepsilon\) follows. The final `linarith` closes the goal after reordering the summands.

The second `IsINE` inequality is proved in the same pattern. Fix \(\eta\in Y\). The diagonal term \(\nu_{\mathrm{val}}k(\bar y,\bar y)\) is rewritten to zero using \(k(\bar y,\bar y)=0\). The remaining goal is
\[
g(\bar y)+\mu_\infty c(\bar x,\bar y)
\le
g(\eta)+\mu_\infty c(\bar x,\eta)+\nu_{\mathrm{val}}k(\bar y,\eta).
\]
Again `le_of_forall_pos_lt_add` reduces it to an \(\varepsilon\)-strict inequality. With \(\varepsilon_0=\varepsilon/5\), lower semicontinuity of \(g\) and the product lemma give
\[
g(\bar y)+\mu_\infty c(\bar x,\bar y)-2\varepsilon_0
<
g(y_{n+1})+\mu_n c(x_{n+1},y_{n+1}).
\]
The algorithm step `alg.step_y n η` gives
\[
g(y_{n+1})+\mu_n c(x_{n+1},y_{n+1})+\nu_n k(y_n,y_{n+1})
\le
g(\eta)+\mu_n c(x_{n+1},\eta)+\nu_n k(y_n,\eta).
\]
Since \(\nu_n k(y_n,y_{n+1})\ge0\), this term may be dropped:
\[
g(y_{n+1})+\mu_n c(x_{n+1},y_{n+1})
\le
g(\eta)+\mu_n c(x_{n+1},\eta)+\nu_n k(y_n,\eta).
\]
Using the upper bound for \(\nu_n k(y_n,\eta)\), together with \(\bar\nu\le\nu_{\mathrm{val}}\) and \(k(\bar y,\eta)\ge0\), and using the convergence of the partial right-hand side, one obtains
\[
g(\eta)+\mu_n c(x_{n+1},\eta)+\nu_n k(y_n,\eta)
<
g(\eta)+\nu_{\mathrm{val}}k(\bar y,\eta)
+\mu_\infty c(\bar x,\eta)+2\varepsilon_0.
\]
Combining the chains gives
\[
g(\bar y)+\mu_\infty c(\bar x,\bar y)
<
g(\eta)+\nu_{\mathrm{val}}k(\bar y,\eta)
+\mu_\infty c(\bar x,\eta)+4\varepsilon_0,
\]
and \(4\varepsilon_0=4\varepsilon/5<\varepsilon\). The final `nlinarith` and `linarith` close the arithmetic.

Finally, the proof constructs the conjunction in the definition of `IsINE`. The first branch introduces an arbitrary \(\xi\), applies the already proved \(x\)-inequality, and uses `simpa [add_comm, add_left_comm, add_assoc]` to match the exact associativity and commutativity of addition used in `IsINE`. The second branch does the same for an arbitrary \(\eta\) with the \(y\)-inequality. The hypotheses `hh_lsc`, `hk_lsc`, `hf_bdd_below`, and `hg_bdd_below` are present in the theorem statement but are not invoked in the displayed proof; the proof needs only the lower semicontinuity of \(f\), \(g\), and \(c\), the separate continuity components used for \(h\), \(k\), and \(c\), and the nonnegativity, diagonal, convergence, and limsup hypotheses listed above.
